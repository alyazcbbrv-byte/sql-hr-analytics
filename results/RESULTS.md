### Q20: Organisation depth (recursive CTE)

**Question:** How many management layers are there and how many people sit on each layer?

```sql
WITH RECURSIVE org (employee_id, manager_id, level) AS (
    SELECT employee_id, manager_id, 0
    FROM employees
    WHERE manager_id IS NULL
    UNION ALL
    SELECT e.employee_id, e.manager_id, org.level + 1
    FROM employees e
    JOIN org ON e.manager_id = org.employee_id
)
SELECT org.level,
       COUNT(*)                   AS employees,
       ROUND(AVG(e.salary), 0)    AS avg_salary
FROM org
JOIN employees e ON e.employee_id = org.employee_id
GROUP BY org.level
ORDER BY org.level;
```

### Q25: Hiring trend with running total and year-over-year change

**Question:** How did the headcount build up year by year? (SUM OVER, LAG)

```sql
WITH yearly AS (
    SELECT CAST(strftime('%Y', hire_date) AS INTEGER) AS hire_year,
           COUNT(*) AS hires
    FROM employees
    GROUP BY hire_year
)
SELECT hire_year,
       hires,
       SUM(hires) OVER (ORDER BY hire_year)                               AS cumulative_headcount,
       hires - LAG(hires) OVER (ORDER BY hire_year)                       AS change_vs_prev_year,
       ROUND(100.0 * (hires - LAG(hires) OVER (ORDER BY hire_year))
             / LAG(hires) OVER (ORDER BY hire_year), 0)                   AS pct_change
FROM yearly
ORDER BY hire_year;
```

### Q27: Salary compression inside a job

**Question:** Do newcomers earn as much as, or more than, the longest-serving colleagues in the same job? (FIRST_VALUE / LAST_VALUE)

```sql
WITH w AS (
    SELECT j.job_title,
           e.hire_date,
           e.salary,
           COUNT(*) OVER (PARTITION BY e.job_id) AS job_size,
           FIRST_VALUE(e.salary) OVER (PARTITION BY e.job_id ORDER BY e.hire_date, e.employee_id
                                       ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS salary_of_longest_serving,
           LAST_VALUE(e.salary)  OVER (PARTITION BY e.job_id ORDER BY e.hire_date, e.employee_id
                                       ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS salary_of_newest_hire,
           FIRST_VALUE(e.hire_date) OVER (PARTITION BY e.job_id ORDER BY e.hire_date, e.employee_id
                                       ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS first_hire
    FROM employees e
    JOIN jobs j ON j.job_id = e.job_id
)
SELECT DISTINCT job_title, job_size,
       salary_of_longest_serving,
       salary_of_newest_hire,
       salary_of_newest_hire - salary_of_longest_serving AS newest_minus_longest,
       CASE WHEN salary_of_newest_hire >= salary_of_longest_serving THEN 'compressed' ELSE 'normal' END AS status
FROM w
WHERE job_size >= 3
ORDER BY newest_minus_longest DESC;
```

### Q30: Tenure buckets and salary

**Question:** Do longer-serving employees earn more? (tenure measured at the snapshot date)

```sql
WITH snap AS (SELECT MAX(hire_date) AS d FROM employees),
t AS (
    SELECT e.salary,
           (julianday(snap.d) - julianday(e.hire_date)) / 365.25 AS tenure_years
    FROM employees e, snap
)

            WHEN tenure_years < 4 THEN '2) 2 - 4 years'
            WHEN tenure_years < 6 THEN '3) 4 - 6 years'

       COUNT(*)                   AS headcount,
       ROUND(AVG(salary), 0)      AS avg_salary,
       ROUND(AVG(tenure_years), 1) AS avg_tenure_years
FROM t
GROUP BY tenure_bucket
```

### Q32: Department scorecard (capstone query)

**Question:** One table for management: size, cost, share of payroll, tenure, commission and location of each department.

```sql
WITH snap AS (SELECT MAX(hire_date) AS d FROM employees),
stats AS (
    SELECT e.department_id,
           COUNT(*)                                                       AS headcount,
           SUM(e.salary)                                                  AS payroll,
           ROUND(AVG(e.salary), 0)                                        AS avg_salary,
           ROUND(AVG((julianday(snap.d) - julianday(e.hire_date)) / 365.25), 1) AS avg_tenure_years,
           SUM(e.commission_pct IS NOT NULL)                              AS on_commission
    FROM employees e, snap
    WHERE e.department_id IS NOT NULL
    GROUP BY e.department_id
)
SELECT d.department_name,
       m.first_name || ' ' || m.last_name AS manager,
       l.city || ', ' || c.country_name   AS location,
       s.headcount,
       s.payroll,
       ROUND(100.0 * s.payroll / SUM(s.payroll) OVER (), 1) AS pct_of_payroll,
       s.avg_salary,
       s.avg_tenure_years,
       s.on_commission
FROM stats s
JOIN departments d     ON d.department_id = s.department_id
LEFT JOIN employees m  ON m.employee_id = d.manager_id
JOIN locations l       ON l.location_id = d.location_id
JOIN countries c       ON c.country_id = l.country_id
ORDER BY s.payroll DESC;
```
